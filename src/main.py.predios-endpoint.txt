from fastapi import APIRouter, Depends, Query, HTTPException
from sqlalchemy.orm import Session, joinedload, selectinload
from sqlalchemy import and_
from typing import Optional

router = APIRouter(prefix="/predios", tags=["predios"])


@router.get("/filtrados", response_model=list[dict])
def get_predios_filtrados(
    db: Session = Depends(get_db),
    rol: Optional[str] = Query(None),
    dept_id: Optional[str] = Query(None),
    skip: int = Query(0),
    limit: int = Query(100),
):
    """
    Obtiene predios filtrados por rol y departamento.
    
    OPTIMIZACIÓN CLAVE:
    - joinedload() precarga Agricultor en la misma consulta
    - selectinload() precarga Departamento y Lotes
    - Una sola query en lugar de N+1
    
    Ejemplo de uso:
    GET /api/core/predios/filtrados?rol=admin&dept_id=15
    """
    query = db.query(Predio)

    # JOINEDLOAD: Carga Agricultor (relación 1:N)
    # Evita una query por cada Predio
    query = query.options(
        joinedload(Predio.agricultor),
        selectinload(Predio.departamento),
        selectinload(Predio.lotes)
    )

    # Filtrar por departamento
    if dept_id:
        query = query.filter(Predio.departamento_id == dept_id)

    # Filtrar por rol
    if rol and rol == "admin":
        # Admin ve todos los predios del departamento
        pass
    elif rol and rol == "tecnico":
        # Técnico solo ve predios asignados
        query = query.filter(Predio.tecnico_asignado_id != None)

@router.get("/optimized-hierarchy", response_model=list[dict])
def get_predios_con_jerarquia(db: Session = Depends(get_db)):
    """
    PILAR 3: Endpoint ultra-optimizado
    Retorna la jerarquía completa en UNA SOLA QUERY:
    Departamento -> Agricultor -> Predio -> Lotes -> Cultivos
    
    Uso desde Angular:
    inspectionService.getInspeccionesOptimizado()
    """
    query = db.query(Predio).options(
        selectinload(Predio.departamento),
        selectinload(Predio.agricultor),
        selectinload(Predio.lotes).selectinload(Lote.cultivo),
        selectinload(Predio.inspecciones).selectinload(Inspeccion.sub_inspecciones)
    )

    predios = query.all()

    return [
        {
            "id": p.id,
            "nombre": p.nombre,
            "departamento": {
                "id": p.departamento.id,
                "nombre": p.departamento.nombre,
            } if p.departamento else None,
            "agricultor": {
                "id": p.agricultor.id,
                "nombre": p.agricultor.nombre,
                "email": p.agricultor.email,
                "telefono": p.agricultor.telefono,
            } if p.agricultor else None,
            "lotes": [
                {
                    "id": l.id,
                    "nombre": l.nombre,
                    "cultivo": l.cultivo.nombre if l.cultivo else None,
                    "area": l.area,
                }
                for l in (p.lotes or [])
            ],
            "inspecciones": [
                {
                    "id": i.id,
                    "estado": i.estado,
                    "fecha": i.fecha_inspeccion,
                    "sub_inspecciones_count": len(i.sub_inspecciones) if i.sub_inspecciones else 0,
                }
                for i in (p.inspecciones or [])
            ],
        }
        for p in predios
    ]


@router.patch("/{predio_id}/asignar-tecnico")
def asignar_tecnico_predio(
    predio_id: str,
    tecnico_id: str,
    db: Session = Depends(get_db)
):
    """
    PILAR 3: Asigna un técnico oficial a un lugar de producción
    Solo Admin puede ejecutar esta acción
    
    Payload:
    {
      "tecnico_id": "uuid-del-tecnico"
    }
    """
    predio = db.query(Predio).filter(Predio.id == predio_id).first()
    if not predio:
        raise HTTPException(status_code=404, detail="Predio no encontrado")

    tecnico = db.query(Usuario).filter(
        and_(
            Usuario.id == tecnico_id,
            Usuario.rol == "tecnico",
            Usuario.activo == True
        )
    ).first()
    if not tecnico:
        raise HTTPException(status_code=404, detail="Técnico no encontrado o inactivo")

    predio.tecnico_asignado_id = tecnico_id
    db.commit()

    return {
        "predio_id": predio_id,
        "tecnico_asignado": {
            "id": tecnico.id,
            "nombre": tecnico.nombre,
            "email": tecnico.email,
        }
    }
